function buildBoard(){ board.innerHTML=''; // Create the coordinate grid and Y-axis numbers. for(let y=10;y>=0;y--){ let yl=document.createElement('div'); yl.className='y-tick'; yl.style.gridColumn='1'; yl.style.gridRow=String(11-y); yl.textContent=y; board.appendChild(yl); for(let x=0;x<=10;x++){ let b=document.createElement('button'); b.type='button'; b.className='cell'; b.dataset.key=x+','+y; b.setAttribute('aria-label',`Search coordinate ${x}, ${y}`); b.style.gridColumn=String(x+2); b.style.gridRow=String(11-y); const lm=landmarkPositions.find(p=>p.key===b.dataset.key); if(lm){ b.classList.add('landmark'); b.textContent=lm.icon; b.title=`${lm.name} at (${x}, ${y})`; b.disabled=true; b.setAttribute('aria-label',`${lm.name} at coordinate ${x}, ${y}`); } else{ b.addEventListener('click',()=>search(x,y,b)); } board.appendChild(b); } } // Blank corner beneath the Y-axis. const corner=document.createElement('div'); corner.style.gridColumn='1'; corner.style.gridRow='12'; board.appendChild(corner); // Add X-axis numbers from 0 to 10. for(let x=0;x<=10;x++){ const xt=document.createElement('div'); xt.className='x-tick'; xt.style.gridColumn=String(x+2); xt.style.gridRow='12'; xt.textContent=x; board.appendChild(xt); } }
